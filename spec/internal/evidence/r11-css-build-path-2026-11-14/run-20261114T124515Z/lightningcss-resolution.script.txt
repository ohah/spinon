import { createRequire } from "node:module"; import { existsSync, readFileSync } from "node:fs"; import { dirname, join } from "node:path"; const require = createRequire(process.cwd() + "/package.json"); const resolved = require.resolve("lightningcss"); let dir = dirname(resolved); while (!existsSync(join(dir, "package.json"))) { const parent = dirname(dir); if (parent === dir) throw new Error("package.json ancestor not found"); dir = parent; } const pkg = JSON.parse(readFileSync(join(dir, "package.json"), "utf8")); console.log(JSON.stringify({ resolvedPath: resolved, packageDir: dir, packageVersion: pkg.version, node: process.version }, null, 2));
